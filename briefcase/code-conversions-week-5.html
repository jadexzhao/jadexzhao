Snippet 2 · csv.DictReader scores.csv · print name

Week 5 · snippet 02

If the embed is blocked, open the shared Drive file: codeconv_w5_snippet02_jadezhao.mp4.

Try it · read the code

import csv

with open("scores.csv", "r", newline="") as file:
    # Scratch: set reader to DictReader (header row becomes the key labels)
    reader = csv.DictReader(file)  # first line = column names · later lines = dicts
    for row in reader:  # each row type: dict (mapping), not a plain list
        # Scratch: say (row's name field)
        print(row["name"])  # predicted: each person's name string, one per line

Snippet 3 · DictReader sum scores · int conversion

import csv

total = 0
with open("scores.csv", "r", newline="") as file:
    reader = csv.DictReader(file)
    for row in reader:
        # Scratch: change total by (score turned into a number)
        total += int(row["score"])  # without int(), "85"+"90" would glue text, not add
print(total)  # predicted: sum of all score numbers in the file

import csv

with open("bookmarks.csv", "w", newline="") as file:
    writer = csv.writer(file)  # helper that joins list items with commas
